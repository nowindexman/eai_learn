## 1. 数据血缘

数据血缘回答的核心问题：**"这个训练数据集里的每一帧数据，最初来自哪个 bag？"**

### 血缘链路

```text
原始 bag (SHA-256: abc...)
  │
  ├─ inspect 阶段
  │  └─ inspect_report.json (工具版本, 配置 v1.2)
  │
  ├─ extract 阶段
  │  └─ intermediate/extracted.jsonl (帧数: 2097)
  │
  ├─ align 阶段 (FPS=30, tolerance=20ms)
  │  └─ intermediate/aligned.parquet (帧数: 2090, 丢弃: 7)
  │
  ├─ quality 阶段
  │  └─ quality_report.json (grade: accept, issues: [...])
  │
  └─ dataset 阶段 (schema: dataset-v2, LeRobot v3)
     └─ data/derived/dataset-v2/episode_000/
        ├── data.parquet
        ├── videos/...
        └── manifest.json (source_sha256: abc..., config: v1.2, ...)
              │
              └─ 训练实验 (exp_20260810)
                 ├── checkpoint.pt
                 └── provenance.json (dataset: v2, commit: def..., seed: 42)
```

每一层都应该能追溯到上一层。如果训练效果不好，你可以从 checkpoint 反查到数据集版本，再反查到原始 bag 和处理配置。

### 血缘记录实现

```python
import hashlib
import json
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path


@dataclass(frozen=True)
class LineageRecord:
    """一个处理步骤的血缘记录。"""
    step: str                          # inspect / extract / align / quality / dataset
    timestamp: str                     # UTC ISO 时间
    inputs: list[dict]                 # 输入文件路径 + SHA-256
    outputs: list[dict]                # 输出文件路径 + SHA-256
    config_hash: str                   # 配置文件的 hash
    code_commit: str                   # Git commit hash
    parameters: dict[str, object] = field(default_factory=dict)  # 关键参数

    def to_json(self) -> str:
        return json.dumps(self.__dict__, indent=2, ensure_ascii=False)


def record_lineage(
    step: str,
    input_paths: list[Path],
    output_paths: list[Path],
    config_path: Path,
    parameters: dict[str, object] | None = None,
) -> LineageRecord:
    """记录一个处理步骤的血缘。"""
    import subprocess

    def file_info(path: Path) -> dict:
        return {
            "path": str(path),
            "sha256": sha256_file(path),
            "size_bytes": path.stat().st_size if path.exists() else 0,
        }

    code_commit = subprocess.check_output(
        ["git", "rev-parse", "HEAD"],
    ).decode().strip()

    return LineageRecord(
        step=step,
        timestamp=datetime.now(timezone.utc).isoformat(),
        inputs=[file_info(p) for p in input_paths],
        outputs=[file_info(p) for p in output_paths],
        config_hash=sha256_file(config_path),
        code_commit=code_commit,
        parameters=parameters or {},
    )
```

## 2. 质量看板

### 需要监控的指标

```text
采集层指标：
  ├── 每日新增 episode 数
  ├── 每日采集时长
  ├── 多传感器同步质量（mean/max delta）
  └── 采集失败率

处理层指标：
  ├── 处理吞吐量（episodes/hour）
  ├── 帧完整性（actual/expected 比率）
  ├── 解码失败率
  ├── 拒绝率（quality grade = reject 的比例）
  └── 处理错误按类型分布

数据集层指标：
  ├── 总 episode 数和总帧数
  ├── 任务分布（各任务 episode 数占比）
  ├── 时间范围覆盖
  ├── stats 漂移（v2 vs v1 的 mean/std 变化）
  └── 存储使用量趋势
```

### 最小看板实现

用简单的 HTML + JSON 数据文件，不需要复杂的基础设施：

```python
"""生成数据集质量看板数据。"""
import json
from datetime import datetime, timezone
from pathlib import Path
from dataclasses import dataclass, field


@dataclass
class DatasetDashboard:
    """数据集质量看板数据。"""
    generated_at: str
    dataset_version: str
    episodes: list[dict] = field(default_factory=list)
    summary: dict = field(default_factory=dict)

    def add_episode(self, manifest_path: Path) -> None:
        manifest = json.loads(manifest_path.read_text())
        quality_path = manifest_path.parent / "quality_report.json"
        quality = json.loads(quality_path.read_text()) if quality_path.exists() else {}

        self.episodes.append({
            "episode_id": manifest["episode_id"],
            "frames": manifest["frames"],
            "duration_seconds": manifest.get("duration_seconds", 0),
            "quality_grade": quality.get("grade", "unknown"),
            "issues": quality.get("issues", []),
        })

    def compute_summary(self) -> None:
        grades = [e["quality_grade"] for e in self.episodes]
        self.summary = {
            "total_episodes": len(self.episodes),
            "total_frames": sum(e["frames"] for e in self.episodes),
            "accepted": grades.count("accept"),
            "warning": grades.count("warning"),
            "rejected": grades.count("reject"),
            "avg_frames": (
                sum(e["frames"] for e in self.episodes) / len(self.episodes)
                if self.episodes else 0
            ),
        }

    def to_json(self, output_path: Path) -> None:
        self.compute_summary()
        output_path.write_text(
            json.dumps(self.__dict__, indent=2, ensure_ascii=False),
        )


# 使用
dashboard = DatasetDashboard(
    generated_at=datetime.now(timezone.utc).isoformat(),
    dataset_version="dataset-v2",
)

for manifest in sorted(Path("data/derived/dataset-v2").rglob("manifest.json")):
    dashboard.add_episode(manifest)

dashboard.to_json(Path("reports/dashboard.json"))
```

生成的 `dashboard.json` 可以用 Grafana、简单的 HTML 页面或 Jupyter Notebook 可视化。

### 数据测试

```python
"""数据质量测试——确保转换结果符合契约。"""
import json
from pathlib import Path

import pandas as pd
import pytest


def test_parquet_has_required_columns(tmp_path: Path) -> None:
    """Parquet 必须包含必需列。"""
    required = {
        "index", "episode_index", "frame_index",
        "timestamp", "observation.state", "action",
    }

    parquet_files = list(tmp_path.rglob("*.parquet"))
    assert parquet_files, "No parquet files found"

    for f in parquet_files:
        df = pd.read_parquet(f)
        missing = required - set(df.columns)
        assert not missing, f"{f.name} missing columns: {missing}"


def test_info_json_is_valid(tmp_path: Path) -> None:
    """info.json 必须有正确的结构和值。"""
    info_path = tmp_path / "meta" / "info.json"
    assert info_path.exists(), "info.json not found"

    info = json.loads(info_path.read_text())
    assert info["codebase_version"].startswith("v3.")
    assert info["fps"] > 0
    assert "observation.state" in info["features"]
    assert "action" in info["features"]


def test_stats_json_covers_all_numeric_features(tmp_path: Path) -> None:
    """stats.json 必须覆盖所有数值特征。"""
    info = json.loads((tmp_path / "meta" / "info.json").read_text())
    stats = json.loads((tmp_path / "meta" / "stats.json").read_text())

    numeric_features = [
        name for name, feat in info["features"].items()
        if feat["dtype"] in ("float32", "float64", "int32", "int64")
    ]

    for feat in numeric_features:
        assert feat in stats, f"stats.json missing feature: {feat}"
        assert "mean" in stats[feat]
        assert "std" in stats[feat]


def test_no_nan_in_state(tmp_path: Path) -> None:
    """状态数据不能有 NaN。"""
    import numpy as np

    for parquet in tmp_path.rglob("*.parquet"):
        df = pd.read_parquet(parquet)
        states = np.array(df["observation.state"].tolist())
        assert not np.isnan(states).any(), f"NaN found in {parquet.name}"


def test_episode_indices_are_sequential(tmp_path: Path) -> None:
    """episode_index 必须从 0 开始连续编号。"""
    for parquet in tmp_path.rglob("*.parquet"):
        df = pd.read_parquet(parquet)
        indices = sorted(df["episode_index"].unique())
        expected = list(range(len(indices)))
        assert indices == expected, f"Non-sequential episodes: {indices}"
```

## 4. 告警

当流水线出问题时，你需要第一时间知道：

| 告警场景 | 触发条件 | 严重级别 | 通知方式 |
|---------|---------|---------|---------|
| 处理失败率飙升 | 失败 episode / 总数 > 10% | CRITICAL | 立即通知 |
| 帧完整性下降 | 平均帧完整性 < 90% | HIGH | 当日汇总 |
| 同步质量恶化 | mean delta > 5ms | HIGH | 当日汇总 |
| 磁盘空间不足 | 剩余 < 10% | CRITICAL | 立即通知 |
| stats 漂移 | 新批次 mean 偏移 > 3σ | MEDIUM | 周报 |
| 流水线超时 | 单 episode 处理 > 30min | MEDIUM | 周报 |

```python
"""简化的告警检查器。"""
import json
import smtplib
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Alert:
    severity: str    # CRITICAL / HIGH / MEDIUM / LOW
    category: str
    message: str
    episode_id: str | None = None


def check_quality_alerts(
    dashboard_path: Path,
    thresholds: dict[str, float],
) -> list[Alert]:
    """检查质量看板数据，生成告警。"""
    dashboard = json.loads(dashboard_path.read_text())
    summary = dashboard["summary"]
    alerts: list[Alert] = []

    total = summary["total_episodes"]
    if total == 0:
        return [Alert("CRITICAL", "no_episodes", "No episodes in dataset")]

    # 拒绝率检查
    reject_rate = summary["rejected"] / total
    if reject_rate > thresholds.get("max_reject_rate", 0.1):
        alerts.append(Alert(
            "HIGH", "reject_rate",
            f"Reject rate {reject_rate:.1%} exceeds threshold "
            f"{thresholds['max_reject_rate']:.1%}",
        ))

    # 检查个别 episode 的异常
    for episode in dashboard["episodes"]:
        if episode["frames"] < thresholds.get("min_frames", 100):
            alerts.append(Alert(
                "MEDIUM", "low_frame_count",
                f"Episode {episode['episode_id']} has only "
                f"{episode['frames']} frames",
                episode["episode_id"],
            ))

    return alerts
```

## 5. 运维检查清单

### 日常检查

```text
□ 流水线是否正常运行（无卡死、无积压）
□ 新增数据的质量是否达标
□ 磁盘空间是否充足
□ 告警是否有未处理的
```

### 周度检查

```text
□ 数据集统计量是否有异常漂移
□ 训练团队是否反馈数据质量问题
□ 是否需要数据集版本升级
□ 存储成本是否在预算内
```

### 月度检查

```text
□ 数据采集覆盖率是否满足任务需求
□ 是否有需要归档或清理的旧数据
□ 流水线性能是否需要优化
□ 文档和 changelog 是否更新
```

## 三阶段完成后的下一步

你现在具备了从采集到训练的完整数据工程能力。后续深入方向：

- **垂直领域**：操作（manipulation）/ 导航（navigation）/ 人形（locomotion）各有不同的数据特点。
- **自动化标注**：用已有模型对新数据进行自动标注，减少人工审核成本。
- **主动学习**：识别模型最需要的数据类型，指导下一轮采集策略。
- **跨设备数据对齐**：不同机器人、不同传感器的数据如何统一到同一训练框架。
- **Sim-to-real 数据混合**：仿真数据和真实数据的配比、筛选和质量评估。

这些专题可以在实际项目中按需深入。进阶计划到这里，你已经从"会读 bag"升级到"能独立负责具身智能数据全链路"了。
